cmake_minimum_required(VERSION 3.20)
project(AndroidEmuCore LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(androidemu_core ImageKit/SparseImage.cpp ImageKit/ZipArchive.cpp Input/Touch.cpp Display/FrameMailbox.cpp ADB/Protocol.cpp ADB/Client.cpp ADB/RSAPublicKey.cpp)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_link_libraries(androidemu_core PUBLIC z)
else()
  find_package(ZLIB REQUIRED)
  target_link_libraries(androidemu_core PUBLIC ZLIB::ZLIB)
endif()
target_include_directories(androidemu_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(androidemu_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(EMU_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(EMU_SANITIZE)
  target_compile_options(androidemu_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(androidemu_core PUBLIC -fsanitize=address,undefined)
endif()
option(EMU_BUILD_TESTS "Build host native tests" ON)
if(EMU_BUILD_TESTS)
  find_package(Threads REQUIRED)
  enable_testing()
  add_executable(zip_extract Tests/ZipExtractTool.cpp)
  target_link_libraries(zip_extract PRIVATE androidemu_core)
  add_executable(image_copy Tests/ImageCopyTool.cpp)
  target_link_libraries(image_copy PRIVATE androidemu_core)
  add_executable(core_tests Tests/CoreTests.cpp)
  target_link_libraries(core_tests PRIVATE androidemu_core Threads::Threads)
  target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME core COMMAND core_tests)
  add_executable(adb_tests Tests/ADBTests.cpp)
  target_link_libraries(adb_tests PRIVATE androidemu_core Threads::Threads)
  target_compile_options(adb_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME adb COMMAND adb_tests)
  add_executable(gpu_transport_tests Tests/GpuTransportTests.cpp ThirdParty/AndroidQemuCompat/qemu/gpu.c)
  target_include_directories(gpu_transport_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(gpu_transport_tests PRIVATE Threads::Threads)
  target_compile_options(gpu_transport_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  if(EMU_SANITIZE)
    target_compile_options(gpu_transport_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(gpu_transport_tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME gpu_transport COMMAND gpu_transport_tests)
  set_tests_properties(gpu_transport PROPERTIES TIMEOUT 15)
  add_executable(gpu_startup_tests Tests/GpuStartupTests.cpp)
  target_include_directories(gpu_startup_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(gpu_startup_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  add_test(NAME gpu_startup COMMAND gpu_startup_tests)
endif()
